곤충은 스스로 체온을 못 만드는 변온동물이라, 주변이 따뜻해야 자라요. 그런데 어느 온도 아래로는 아예 자라지 않아요. 이 기준 온도를 발육영점이라고 해요. 하루 평균기온이 발육영점보다 높은 만큼을 매일 저금한다고 생각해 보세요. 그 저금이 정해진 목표(유효적산온도)에 닿는 날 알이 깨거나 성충이 나와요. 그래서 따뜻한 봄엔 벌레가 일찍 나오고, 방제 날짜도 앞당겨야 해요.

올봄 날씨

예시 해충 값이에요. 실제 값은 해충과 발육 단계마다 달라요.

let yr='normal'; const st=makeCanvas('cv',0.95,draw); const OFF={cool:-1.8,normal:0,warm:2.2}; function temps(type){ // 3월 1일부터 150일 일평균기온 const out=[]; let seed=7; for(let d=0;d<150;d++){ seed=(seed*9301+49297)%233280; const noise=(seed/233280-0.5)*4; const t = 5 + 20*Math.sin(Math.PI*clamp(d/170,0,1)) + OFF[type] + noise; out.push(t); } return out; } function emerge(type){ const T=temps(type), b=val('base'), K=val('K'); let c=0, cum=[]; for(let d=0;d=K && cum.found===undefined) cum.found=d; } return {T,cum,day:cum.found}; } function dstr(d){ if(d===undefined) return '7월 말 이후'; const t=new Date(2026,2,1+d); return (t.getMonth()+1)+'월 '+t.getDate()+'일'; } function draw(){ const {ctx,W,H}=st; ctx.clearRect(0,0,W,H); const r=emerge(yr), b=val('base'), K=val('K'); const L=W*0.1, R=W*0.95, T1=H*0.06, B1=H*0.46, T2=H*0.55, B2=H*0.9; const X=d=>L+(R-L)*d/150, Yt=t=>B1-(B1-T1)*(t+2)/32, Yc=c=>B2-(B2-T2)*Math.min(c,700)/700; // 위: 일평균기온 line(ctx,L,B1,R,B1,CH.chalk,1.5); line(ctx,L,T1,L,B1,CH.chalk,1.5); txt(ctx,'일평균기온',L+4,T1+12,15,CH.dim,'left',W); ctx.beginPath(); r.T.forEach((t,d)=>{ const x=X(d),y=Yt(t); d?ctx.lineTo(x,y):ctx.moveTo(x,y); }); ctx.strokeStyle=CH.orange; ctx.lineWidth=2; ctx.stroke(); // 발육영점 위 면적 칠하기 const upto = r.day===undefined? r.T.length-1 : r.day; for(let d=0;d<=upto;d++){ const t=r.T[d]; if(t>b) rect(ctx,X(d),Yt(t),Math.max(1,(R-L)/150),Yt(b)-Yt(t),'rgba(244,211,94,.35)'); } line(ctx,L,Yt(b),R,Yt(b),CH.blue,2,[6,4]); txt(ctx,'발육영점 '+b+'℃',R,Yt(b)-5,15,CH.blue,'right',W); // 아래: 누적 line(ctx,L,B2,R,B2,CH.chalk,1.5); line(ctx,L,T2,L,B2,CH.chalk,1.5); txt(ctx,'쌓인 온도(일도)',L+4,T2+12,15,CH.dim,'left',W); ctx.beginPath(); r.cum.forEach((c,d)=>{ const x=X(d),y=Yc(c); d?ctx.lineTo(x,y):ctx.moveTo(x,y); }); ctx.strokeStyle=CH.yellow; ctx.lineWidth=2.5; ctx.stroke(); line(ctx,L,Yc(K),R,Yc(K),CH.pink,2,[6,4]); txt(ctx,'목표 '+K,L+6,Yc(K)-5,15,CH.pink,'left',W); if(r.day!==undefined){ line(ctx,X(r.day),T1,X(r.day),B2,CH.green,2); txt(ctx,'발생!',X(r.day)+4,T2-4,17,CH.green,'left',W); } ['3월','4월','5월','6월','7월'].forEach((m,i)=>txt(ctx,m,X(i*30.5+2),H-4,14,CH.dim,'left',W)); } function update(){ setText('vB',val('base')+'℃'); setText('vK',val('K')+' 일도(℃·일)'); const r=emerge(yr), n=emerge('normal'); setText('sDay',dstr(r.day)); setText('sCtl', r.day===undefined?'-':dstr(r.day)+' ~ '+dstr(r.day+7)); const diff = (r.day!==undefined&&n.day!==undefined)? r.day-n.day : null; setText('sDiff', diff===null?'-': diff===0?'평년과 같음': diff<0? Math.abs(diff)+'일 빠름' : diff+'일 늦음'); if(yr==='warm') msg('따뜻한 봄: 온도 저금이 빨리 쌓여 발생이 앞당겨져요 → 방제도 앞당겨요','#f0a868'); else if(yr==='cool') msg('서늘한 봄: 발생이 늦어져요. 달력만 보고 약을 치면 너무 일러요','#9ed2f2'); else msg('노란 면적(발육영점 위 온도)이 매일 쌓여 목표에 닿으면 발생','#a8e0a0'); draw(); } segButtons('yr',v=>{ yr=v; update(); }); onInputs(['base','K'],update);
  1. 발육영점(발육 최저온도): 그 온도 이하에서는 발육하지 않는 기준 온도. 많은 곤충이 대략 8~12℃ 사이예요.
  2. 유효적산온도 = Σ(일평균기온 − 발육영점). 일평균기온이 발육영점보다 낮은 날은 0으로 계산해요. 단위는 일도(degree-day, ℃·일).
  3. 유효적산온도 법칙: K = (T − T₀) × D. 온도가 높으면(T↑) 발육 기간(D)이 짧아져요.
  4. 쓰임: 해충 발생 시기 예찰 → 방제 적기 결정(예: 알에서 깨어난 직후 어린 유충기). 기후변화로 봄이 따뜻해지면 발생이 앞당겨지고 연간 세대 수가 늘 수 있어요.
  5. 계산 연습: 발육영점 10℃, 일평균기온 20℃가 25일 계속되면 (20−10)×25 = 250 일도.
  6. 함께 쓰는 예찰 도구: 페로몬 트랩, 끈끈이 트랩, 유아등(빛 트랩), 표본 조사.
